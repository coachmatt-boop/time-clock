const EMPLOYEE_SHEET = 'Employees'; const LOG_SHEET = 'ClockingLog'; const CONFIG_SHEET = 'Config'; /** * V3 JSON API. * * GET /exec?action=clock&token=TOKEN * GET /exec?action=whosin * GET /exec * * JSONP is used for the external GitHub Pages kiosk so the kiosk page * can stay open while the Apps Script backend records the transaction. */ function doGet(e) { const action = String((e && e.parameter && e.parameter.action) || '').toLowerCase(); const callback = String((e && e.parameter && e.parameter.callback) || '').trim(); if (action === 'clock') { const token = String(e.parameter.token || '').trim(); const result = clockEmployee_(token); return jsonp_(result, callback); } if (action === 'whosin') { return jsonp_(getWhosIn_(), callback); } return ContentService .createTextOutput(JSON.stringify({ success: true, message: 'Time & Attendance API is running', version: 'V3' })) .setMimeType(ContentService.MimeType.JSON); } function clockEmployee_(token) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const employees = ss.getSheetByName(EMPLOYEE_SHEET); const log = ss.getSheetByName(LOG_SHEET); if (!employees || !log) { return result_(false, '', 'SYSTEM ERROR', new Date(), ''); } const cfg = getConfig_(); const kioskName = cfg.KioskName || 'Workshop'; const lockoutSeconds = Number(cfg.DoubleScanLockoutSeconds || 30); if (!token) { return result_(false, '', 'INVALID PASS', new Date(), ''); } const data = employees.getDataRange().getValues(); const now = new Date(); for (let i = 1; i < data.length; i++) { const employeeID = String(data[i][0] || '').trim(); const name = String(data[i][1] || '').trim(); const qrToken = String(data[i][2] || '').trim(); const active = data[i][3]; const currentStatus = String(data[i][4] || 'OUT').toUpperCase(); const lastClock = data[i][5]; if (qrToken === token && isTrue_(active)) { if (!employeeID || !name) { return result_(false, name, 'EMPLOYEE RECORD INCOMPLETE', now, ''); } if (lastClock instanceof Date && !isNaN(lastClock)) { const secondsSinceLast = (now.getTime() - lastClock.getTime()) / 1000; if (secondsSinceLast >= 0 && secondsSinceLast < lockoutSeconds) { return result_(false, name, 'ALREADY CLOCKED', now, currentStatus); } } const newStatus = currentStatus === 'IN' ? 'OUT' : 'IN'; employees.getRange(i + 1, 5).setValue(newStatus); employees.getRange(i + 1, 6).setValue(now); log.appendRow([ now, employeeID, name, newStatus, kioskName ]); SpreadsheetApp.flush(); return result_( true, name, newStatus === 'IN' ? 'CLOCKED IN' : 'CLOCKED OUT', now, newStatus ); } } return result_(false, '', 'PASS NOT RECOGNISED', now, ''); } finally { lock.releaseLock(); } } function generateMissingTokens() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(EMPLOYEE_SHEET); if (!sheet) throw new Error('Employees sheet not found.'); const data = sheet.getDataRange().getValues(); const existing = new Set( data.slice(1) .map(r => String(r[2] || '').trim()) .filter(Boolean) ); let generated = 0; for (let i = 1; i < data.length; i++) { const employeeID = String(data[i][0] || '').trim(); const name = String(data[i][1] || '').trim(); const token = String(data[i][2] || '').trim(); if (employeeID && name && !token) { let newToken; do { newToken = Utilities.getUuid().replace(/-/g, ''); } while (existing.has(newToken)); sheet.getRange(i + 1, 3).setValue(newToken); existing.add(newToken); generated++; } } SpreadsheetApp.flush(); return generated; } function getWhosIn_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(EMPLOYEE_SHEET); if (!sheet) throw new Error('Employees sheet not found.'); const data = sheet.getDataRange().getValues(); const people = []; for (let i = 1; i < data.length; i++) { if (isTrue_(data[i][3]) && String(data[i][4] || '').toUpperCase() === 'IN') { people.push({ employeeID: String(data[i][0] || ''), name: String(data[i][1] || ''), lastClock: data[i][5] instanceof Date ? formatDateTime_(data[i][5]) : '' }); } } people.sort((a, b) => a.name.localeCompare(b.name)); return { success: true, count: people.length, people: people, refreshed: formatDateTime_(new Date()) }; } function resetAllToOut() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(EMPLOYEE_SHEET); if (!sheet) throw new Error('Employees sheet not found.'); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (isTrue_(data[i][3])) { sheet.getRange(i + 1, 5).setValue('OUT'); } } SpreadsheetApp.flush(); } function jsonp_(obj, callback) { const json = JSON.stringify(obj); // Sanitize callback to a safe JS identifier path. if (callback && /^[A-Za-z_$][A-Za-z0-9_$]*(?:\.[A-Za-z_$][A-Za-z0-9_$]*)*$/.test(callback)) { return ContentService .createTextOutput(callback + '(' + json + ');') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } function getConfig_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(CONFIG_SHEET); if (!sheet) return {}; const values = sheet.getDataRange().getValues(); const cfg = {}; for (let i = 1; i < values.length; i++) { const key = String(values[i][0] || '').trim(); if (key) cfg[key] = values[i][1]; } return cfg; } function isTrue_(value) { return value === true || String(value).toUpperCase() === 'TRUE' || String(value).toUpperCase() === 'YES'; } function result_(success, name, message, when, status) { return { success: success, name: name || '', message: message || '', time: formatTime_(when), dateTime: formatDateTime_(when), status: status || '' }; } function formatTime_(date) { return Utilities.formatDate(date, Session.getScriptTimeZone(), 'HH:mm'); } function formatDateTime_(date) { return Utilities.formatDate(date, Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm:ss'); }
